  class SectionTriggerEngine:
      """Determines which template sections to activate based on decisions."""

      def get_active_sections(self, decisions: Decisions, tags: dict) -> List[str]:
          sections = ["appearance", "expression", "lighting", "composition"]  # Always active

          # Body details: enabled for MS+ shots or NSFW
          if decisions.shot_type in [ShotType.MS, ShotType.MFS, ShotType.FS, ...]:
              sections.append("body")
          if decisions.nsfw_level in [NSFWLevel.SUGGESTIVE, NSFWLevel.EXPLICIT]:
              sections.append("body")

          # Pose: enabled for MCU+ shots
          if decisions.shot_type in [ShotType.MCU, ShotType.MS, ...]:
              sections.append("pose")

          # Environment: enabled unless simple background
          if not self._is_simple_background(tags):
              sections.append("environment")

          # Clothing: enabled if clothing tags present
          if self._has_clothing_tags(tags):
              sections.append("clothing")

          return sections

  class StandardMode(BaseMode):
      def __init__(self):
          self.tagger_runner = TaggerRunner()
          self.decision_engine = DecisionEngine()
          self.template_loader = TemplateLoader()
          self.template_builder = TemplateBuilder()

      @property
      def name(self) -> str:
          return "standard"

      def execute(self, image, llm_model, **kwargs) -> GeneratorResult:
          # Step 1: Run all taggers
          tagger_results = self.tagger_runner.run_all(image)

          # Step 2: Make decisions based on tags
          decisions = self.decision_engine.make_decisions(
              tags=self._extract_tags(tagger_results),
              analyzers=self._extract_analyzers(tagger_results)
          )

          # Step 3: Build template prompt from decisions
          template_prompt = self.template_builder.build(
              decisions=decisions,
              templates=self.template_loader.load_templates()
          )

          # Step 4: Call LLM with template prompt
          llm_response = self._call_llm(image, llm_model, template_prompt)

          return GeneratorResult(
              prompt=llm_response,
              metadata={
                  "mode": "standard",
                  "source": "llm_template",
                  "decisions": asdict(decisions),
                  "taggers": tagger_results,
                  "template": {
                      "selected": decisions.subject_type.value,
                      "sections_used": decisions.template_sections,
                  },
                  # ... other metadata
              }
          )

## Phase 10: Node Integration

**Goal**: Wire everything together in the ComfyUI node.

- [ ] 10.1 Complete `generator.py` with mode routing
  ```python
  class SID_PromptGenerator:
      def __init__(self):
          self.modes = {
              "override": OverrideMode(),
              "florence": FlorenceMode(),
              "quick": QuickMode(),
              "standard": StandardMode(),
              "detailed": DetailedMode(),
              "extreme": ExtremeMode(),
          }

      def generate(self, image, llm_model=None, user_prompt=None, analysis_mode="Standard"):
          import json
          import time

          start_time = time.time()

          # Override mode
          if user_prompt is not None and user_prompt.strip():
              result = self.modes["override"].execute(user_prompt=user_prompt)
              return self._format_output(result, start_time)

          # Florence mode
          if llm_model is None:
              result = self.modes["florence"].execute(image=image)
              return self._format_output(result, start_time)

          # LLM modes
          mode_key = analysis_mode.lower()
          result = self.modes[mode_key].execute(
              image=image,
              llm_model=llm_model
          )
          return self._format_output(result, start_time)

      def _format_output(self, result: GeneratorResult, start_time: float):
          result.metadata["timing"] = {
              "total_ms": int((time.time() - start_time) * 1000)
          }
          return (result.prompt, json.dumps(result.metadata, indent=2))
  ```
